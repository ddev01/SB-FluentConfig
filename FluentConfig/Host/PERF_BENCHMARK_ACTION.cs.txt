using FluentConfig;
using System;

public class CPHInline
{
    private const string Title = "FluentConfig Perf Benchmark";
    private const string Version = "1.0";

    public bool Execute()
    {
        try
        {
            Fc.SetLogCallback(msg => CPH.LogInfo("[FC] " + msg));

            // Wall-clock marker before Fc.Open (pairs with FluentConfig_PerfLast).
            CPH.LogInfo("[MENU] trigger " + Title);

            Fc.Open(CPH, Title, Version, ui => ui
                .Section("Benchmark", "benchmark", s => s
                    .Intro("Minimal FluentConfig surface for cold/warm open timing. Close this window between harness runs."))
            );

            return true;
        }
        catch (Exception ex)
        {
            CPH.LogError("FluentConfig Perf Benchmark FAILED: " + ex);
            return false;
        }
    }
}
