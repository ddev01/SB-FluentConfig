using FluentConfig;
using System;

public class CPHInline
{
    public bool Execute()
    {
        try
        {
            if (FluentConfig.FluentConfig.AlreadyOpened("FluentConfig WebView2 Spike", "0.1"))
                return true;

            FluentConfigUi.Create(CPH, "FluentConfig WebView2 Spike", "0.1")
                .Section("Spike", "spike", s => s
                    .Intro("FluentConfig WebView2 host is running."))
                .Show();

            return true;
        }
        catch (Exception ex)
        {
            CPH.LogError("FluentConfig WebView2 spike FAILED: " + ex);
            return false;
        }
    }
}
